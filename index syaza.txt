    <!-- App Logic & Charting Script -->
    <script>
        // Core Mathematical Formulas
        function calculateFx(x) {
            return 30 / (1 + Math.exp(-0.35 * (x - 8)));
        }

        function calculateDerivFx(x) {
            const numerator = 10.5 * Math.exp(-0.35 * (x - 8));
            const denominator = Math.pow(1 + Math.exp(-0.35 * (x - 8)), 2);
            return numerator / denominator;
        }

        // Generate Dataset for Age 0 to 25
        const yearsData = [];
        const yieldData = [];
        for (let x = 0; x <= 25; x += 0.5) {
            yearsData.push(x);
            yieldData.push(calculateFx(x).toFixed(2));
        }

        // Chart.js Configuration
        const ctx = document.getElementById('yieldChart').getContext('2d');
        
        // Gradient background for the curve fill
        const chartGradient = ctx.createLinearGradient(0, 0, 0, 300);
        chartGradient.addColorStop(0, 'rgba(16, 185, 129, 0.4)');
        chartGradient.addColorStop(1, 'rgba(16, 185, 129, 0.0)');

        const yieldChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: yearsData,
                datasets: [{
                    label: 'Cumulative FFB Yield (f(x))',
                    data: yieldData,
                    borderColor: '#10b981',
                    borderWidth: 3,
                    backgroundColor: chartGradient,
                    fill: true,
                    tension: 0.3,
                    pointRadius: 0,
                    pointHoverRadius: 7,
                    pointHoverBackgroundColor: '#f59e0b',
                    pointHoverBorderColor: '#ffffff',
                    pointHoverBorderWidth: 2
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: { display: false },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                return ` Yield: ${context.parsed.y} tonnes/ha`;
                            }
                        }
                    }
                },
                scales: {
                    x: {
                        title: { display: true, text: 'Age of Oil Palm Trees (Years)', font: { weight: 'bold' } },
                        grid: { display: false }
                    },
                    y: {
                        title: { display: true, text: 'Cumulative Yield (Tonnes/Hectare)', font: { weight: 'bold' } },
                        min: 0,
                        max: 35
                    }
                },
                // Trigger action when user clicks anywhere on the line graph
                onClick: (e, elements) => {
                    if (elements.length > 0) {
                        const index = elements[0].index;
                        const selectedYear = yearsData[index];
                        updateAnalysisPanel(selectedYear);
                    }
                },
                // Subtle intro animation
                animation: {
                    duration: 2000,
                    easing: 'easeOutQuart'
                }
            }
        });

        // Update Dashboard Content and Layman Logic
        function updateAnalysisPanel(year) {
            const fX = calculateFx(year);
            const derivFX = calculateDerivFx(year);

            // UI Elements
            const panel = document.getElementById('analysisPanel');
            const placeholder = document.getElementById('placeholderText');
            const content = document.getElementById('analysisContent');

            // Toggle views
            placeholder.classList.add('hidden');
            content.classList.remove('hidden');

            // Insert numbers
            document.getElementById('outYear').innerText = year;
            document.getElementById('outYield').innerText = fX.toFixed(2);
            document.getElementById('outRate').innerText = derivFX.toFixed(2);

            // Generate contextual bio-interpretations for Malaysian Agrotech Context
            let explanation = "";
            let panelBorderColor = "border-slate-500";

            if (year < 4) {
                explanation = `At Year ${year}, the oil palm trees are in their immature, juvenile phase. The low derivative value (${derivFX.toFixed(2)}) confirms that fruit bunch production is minimal as the plant prioritizes root architecture and vegetative growth. Direct commercial harvesting has not effectively started yet.`;
                panelBorderColor = "border-amber-400";
            } else if (year >= 4 && year <= 9) {
                explanation = `This is the prime explosive vegetative growth phase! The derivative is climbing rapidly, meaning your crop output velocity accelerates every single month. At Year ${year}, the plantation is gaining yield momentum quickly—ensure optimal nitrogen-phosphorus-potassium (NPK) fertilization to feed this intense production velocity.`;
                panelBorderColor = "border-emerald-400";
            } else if (year > 9 && year <= 12) {
                explanation = `The graph has crossed its inflection point (around Year 8). While total lifetime crop output is still rising, the rate of change is beginning to slow down from its all-time peak velocity. The tree has transitioned into mature status and stabilized its physiological output.`;
                panelBorderColor = "border-blue-400";
            } else {
                explanation = `At Year ${year}, the palms are in their senior phase. The flat cumulative curve shows that the annual growth velocity ($f'(x)$) has declined substantially. In professional Malaysian estates, when this rate dips near or below baseline operational costs, it alerts management to schedule a complete clear-fell and execute the next 25-year replanting scheme.`;
                panelBorderColor = "border-red-400";
            }

            // Apply dynamic border change based on biological cycle
            panel.className = `bg-slate-900 text-white p-6 rounded-xl shadow-md transition-all duration-300 border-l-8 ${panelBorderColor}`;
            document.getElementById('laymanExplanation').innerText = explanation;
        }
    </script>
